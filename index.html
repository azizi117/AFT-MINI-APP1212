require("dotenv").config(); const express = require("express"); const crypto = require("crypto"); const { Pool } = require("pg"); const rateLimit = require("express-rate-limit"); const helmet = require("helmet"); const app = express(); const pool = new Pool({ connectionString: process.env.DATABASE_URL, ssl: process.env.DATABASE_SSL === "true" ? { rejectUnauthorized: false } : false }); const PORT = process.env.PORT || 3000; const BOT_TOKEN = process.env.BOT_TOKEN; const BOT_USERNAME = process.env.BOT_USERNAME || ""; const DURATION = Math.max(60, Number(process.env.MINING_DURATION_SECONDS || 3600)); const REWARD = Math.max(0, Number(process.env.MINING_REWARD || 40)); const REFERRAL_REWARD = 10; app.use(helmet({ contentSecurityPolicy: false })); app.use(express.json({ limit: "20kb" })); app.use("/api", rateLimit({ windowMs: 60000, limit: 100 })); // DATABASE async function initDB() { await pool.query(` CREATE TABLE IF NOT EXISTS users ( id BIGSERIAL PRIMARY KEY, telegram_id BIGINT UNIQUE NOT NULL, username TEXT, first_name TEXT, balance NUMERIC(20,4) DEFAULT 0, total_mined NUMERIC(20,4) DEFAULT 0, level INTEGER DEFAULT 1, referral_code TEXT UNIQUE NOT NULL, referred_by BIGINT REFERENCES users(telegram_id), created_at TIMESTAMPTZ DEFAULT NOW() ); CREATE TABLE IF NOT EXISTS mining_sessions ( id BIGSERIAL PRIMARY KEY, telegram_id BIGINT REFERENCES users(telegram_id), started_at TIMESTAMPTZ NOT NULL, ends_at TIMESTAMPTZ NOT NULL, claimed BOOLEAN DEFAULT FALSE ); CREATE TABLE IF NOT EXISTS referrals ( id BIGSERIAL PRIMARY KEY, inviter_id BIGINT REFERENCES users(telegram_id), invited_id BIGINT UNIQUE REFERENCES users(telegram_id), reward NUMERIC(20,4) DEFAULT 10, created_at TIMESTAMPTZ DEFAULT NOW() ); CREATE TABLE IF NOT EXISTS missions ( id SERIAL PRIMARY KEY, title TEXT NOT NULL, reward NUMERIC(20,4) NOT NULL, active BOOLEAN DEFAULT TRUE ); CREATE TABLE IF NOT EXISTS user_missions ( id BIGSERIAL PRIMARY KEY, telegram_id BIGINT REFERENCES users(telegram_id), mission_id INTEGER REFERENCES missions(id), completed BOOLEAN DEFAULT FALSE, UNIQUE(telegram_id, mission_id) ); CREATE TABLE IF NOT EXISTS ledger ( id BIGSERIAL PRIMARY KEY, telegram_id BIGINT REFERENCES users(telegram_id), type TEXT NOT NULL, amount NUMERIC(20,4) NOT NULL, description TEXT, created_at TIMESTAMPTZ DEFAULT NOW() ); `); await pool.query(` INSERT INTO missions(title, reward) SELECT 'Welcome to Emerald', 20 WHERE NOT EXISTS ( SELECT 1 FROM missions WHERE title = 'Welcome to Emerald' ); INSERT INTO missions(title, reward) SELECT 'Explore Emerald', 10 WHERE NOT EXISTS ( SELECT 1 FROM missions WHERE title = 'Explore Emerald' ); `); } // TELEGRAM AUTH function verifyTelegram(initData) { if (!BOT_TOKEN || !initData) return null; const params = new URLSearchParams(initData); const hash = params.get("hash"); if (!hash) return null; params.delete("hash"); const check = [...params.entries()] .sort(([a], [b]) => a.localeCompare(b)) .map(([k, v]) => `${k}=${v}`) .join("\n"); const secret = crypto .createHmac("sha256", "WebAppData") .update(BOT_TOKEN) .digest(); const calculated = crypto .createHmac("sha256", secret) .update(check) .digest("hex"); const a = Buffer.from(calculated, "hex"); const b = Buffer.from(hash, "hex"); if (a.length !== b.length || !crypto.timingSafeEqual(a, b)) return null; const authDate = Number(params.get("auth_date")); if (!authDate || Date.now() / 1000 - authDate > 86400) return null; try { return JSON.parse(params.get("user")); } catch { return null; } } async function auth(req, res, next) { try { const tg = verifyTelegram( req.headers["x-telegram-init-data"] ); if (!tg || !tg.id) return res.status(401).json({ error: "Invalid Telegram authentication" }); const referral = String(req.headers["x-referral-code"] || "") .slice(0, 40); const code = crypto.randomBytes(8).toString("hex"); const result = await pool.query(` INSERT INTO users (telegram_id, username, first_name, referral_code) VALUES ($1, $2, $3, $4) ON CONFLICT (telegram_id) DO UPDATE SET username = EXCLUDED.username, first_name = EXCLUDED.first_name RETURNING * `, [ tg.id, tg.username || null, tg.first_name || "Player", code ]); req.user = result.rows[0]; // Apply a referral only once, to a genuinely new account. if (referral && referral !== req.user.referral_code) { const inviter = await pool.query( "SELECT telegram_id FROM users WHERE referral_code=$1", [referral] ); if (inviter.rowCount && String(inviter.rows[0].telegram_id) !== String(tg.id)) { const client = await pool.connect(); try { await client.query("BEGIN"); const changed = await client.query(` UPDATE users SET referred_by=$1 WHERE telegram_id=$2 AND referred_by IS NULL RETURNING telegram_id `, [inviter.rows[0].telegram_id, tg.id]); if (changed.rowCount) { await client.query(` INSERT INTO referrals(inviter_id, invited_id, reward) VALUES ($1, $2, $3) ON CONFLICT (invited_id) DO NOTHING `, [inviter.rows[0].telegram_id, tg.id, REFERRAL_REWARD]); await client.query(` UPDATE users SET balance=balance+$1 WHERE telegram_id=$2 `, [REFERRAL_REWARD, inviter.rows[0].telegram_id]); await client.query(` INSERT INTO ledger(telegram_id,type,amount,description) VALUES ($1,'referral',$2,'Referral reward') `, [inviter.rows[0].telegram_id, REFERRAL_REWARD]); } await client.query("COMMIT"); } catch (e) { await client.query("ROLLBACK"); throw e; } finally { client.release(); } } } next(); } catch (e) { next(e); } } // USER app.get("/api/me", auth, async (req, res, next) => { try { const r = await pool.query(` SELECT telegram_id, username, first_name, balance, total_mined, level, referral_code, created_at FROM users WHERE telegram_id=$1 `, [req.user.telegram_id]); res.json(r.rows[0]); } catch (e) { next(e); } }); // MINING STATUS app.get("/api/mining", auth, async (req, res, next) => { try { const r = await pool.query(` SELECT * FROM mining_sessions WHERE telegram_id=$1 AND claimed=FALSE ORDER BY id DESC LIMIT 1 `, [req.user.telegram_id]); res.json({ session: r.rows[0] || null, duration: DURATION, reward: REWARD }); } catch (e) { next(e); } }); // START MINING app.post("/api/mining/start", auth, async (req, res, next) => { const c = await pool.connect(); try { await c.query("BEGIN"); const active = await c.query(` SELECT id FROM mining_sessions WHERE telegram_id=$1 AND claimed=FALSE AND ends_at>NOW() FOR UPDATE `, [req.user.telegram_id]); if (active.rowCount) { await c.query("ROLLBACK"); return res.status(409).json({ error: "Mining already active" }); } const r = await c.query(` INSERT INTO mining_sessions (telegram_id,started_at,ends_at) VALUES ($1,NOW(),NOW()+($2 * INTERVAL '1 second')) RETURNING * `, [req.user.telegram_id, DURATION]); await c.query("COMMIT"); res.json({ session: r.rows[0] }); } catch (e) { await c.query("ROLLBACK"); next(e); } finally { c.release(); } }); // CLAIM MINING app.post("/api/mining/claim", auth, async (req, res, next) => { const c = await pool.connect(); try { await c.query("BEGIN"); const r = await c.query(` SELECT * FROM mining_sessions WHERE telegram_id=$1 AND claimed=FALSE ORDER BY id DESC LIMIT 1 FOR UPDATE `, [req.user.telegram_id]); const session = r.rows[0]; if (!session) { await c.query("ROLLBACK"); return res.status(404).json({ error: "No session" }); } if (new Date(session.ends_at) > new Date()) { await c.query("ROLLBACK"); return res.status(400).json({ error: "Mining not finished" }); } await c.query( "UPDATE mining_sessions SET claimed=TRUE WHERE id=$1", [session.id] ); await c.query(` UPDATE users SET balance=balance+$1,total_mined=total_mined+$1 WHERE telegram_id=$2 `, [REWARD, req.user.telegram_id]); await c.query(` INSERT INTO ledger(telegram_id,type,amount,description) VALUES ($1,'mining',$2,'Mining reward') `, [req.user.telegram_id, REWARD]); await c.query("COMMIT"); res.json({ success: true, reward: REWARD }); } catch (e) { await c.query("ROLLBACK"); next(e); } finally { c.release(); } }); // REFERRALS app.get("/api/referrals", auth, async (req, res, next) => { try { const r = await pool.query(` SELECT COUNT(*) AS total, COALESCE(SUM(reward),0) AS rewards FROM referrals WHERE inviter_id=$1 `, [req.user.telegram_id]); res.json({ ...r.rows[0], code: req.user.referral_code, link: BOT_USERNAME ? `https://t.me/${BOT_USERNAME}?startapp=${req.user.referral_code}` : null }); } catch (e) { next(e); } }); // MISSIONS app.get("/api/missions", auth, async (req, res, next) => { try { const r = await pool.query(` SELECT m.id,m.title,m.reward, COALESCE(um.completed,FALSE) AS completed FROM missions m LEFT JOIN user_missions um ON um.mission_id=m.id AND um.telegram_id=$1 WHERE m.active=TRUE ORDER BY m.id `, [req.user.telegram_id]); res.json(r.rows); } catch (e) { next(e); } }); app.post("/api/missions/:id/claim", auth, async (req, res, next) => { const c = await pool.connect(); try { await c.query("BEGIN"); const m = await c.query( "SELECT * FROM missions WHERE id=$1 AND active=TRUE", [req.params.id] ); if (!m.rowCount) { await c.query("ROLLBACK"); return res.status(404).json({ error: "Mission not found" }); } const done = await c.query(` INSERT INTO user_missions(telegram_id,mission_id,completed) VALUES ($1,$2,TRUE) ON CONFLICT (telegram_id,mission_id) DO NOTHING RETURNING id `, [req.user.telegram_id, m.rows[0].id]); if (!done.rowCount) { await c.query("ROLLBACK"); return res.status(409).json({ error: "Already claimed" }); } const reward = Number(m.rows[0].reward); await c.query( "UPDATE users SET balance=balance+$1 WHERE telegram_id=$2", [reward, req.user.telegram_id] ); await c.query(` INSERT INTO ledger(telegram_id,type,amount,description) VALUES ($1,'mission',$2,$3) `, [req.user.telegram_id, reward, m.rows[0].title]); await c.query("COMMIT"); res.json({ success: true, reward });
